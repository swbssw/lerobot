# 速度上限:JAKA 跟随主臂的最大角速度,单位 度/秒(保持队友验证的最快安全设置)

# 加速度上限,单位 度/秒^2

# 加加速度上限,单位 度/秒^3

# 主臂角度软死区,单位 度;顺序 J1 J2 J3 J4 J5 J6

# 刷新率,单位 Hz
fps = 125

# 关节方向:J1..J6 顺序,-1=反向、1=正向(当前:J1 和 J5 反向)
joint-signs = -1 1 1 1 -1 1
